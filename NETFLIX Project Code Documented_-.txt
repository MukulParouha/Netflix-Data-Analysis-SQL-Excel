1. Subscription & Revenue Analysis :-        


• What is the monthly subscriber growth trend?
 code:-    
     SELECT 
         DATE_TRUNC('month', start_date::DATE)::DATE AS month_start,
         COUNT(DISTINCT customer_id) AS sub_count
     FROM subscriptions
     WHERE start_date IS NOT NULL AND start_date <> ''
     GROUP BY month_start
    ORDER BY month_start ASC;


• What is the Monthly Recurring Revenue (MRR)?
Code:-
  SELECT 
     DATE_TRUNC('month', start_date::DATE)::DATE AS month_start,
     SUM(CAST(monthly_price AS int)) AS Revenue
  FROM subscriptions
  WHERE start_date IS NOT NULL AND start_date <> '' AND LOWER(status) = 'active'
  GROUP BY month_start
  ORDER BY month_start ASC;


• Which subscription plan generates the highest revenue?
Code:-
    SELECT 
    upper(plan) as plan,
    SUM(CAST(monthly_price AS int)) AS Revenue
FROM subscriptions
WHERE start_date IS NOT NULL AND start_date <> '' AND LOWER(status) = 'active'
GROUP BY 1
ORDER BY 1 ASC;


• Which plan has the highest number of subscribers?
Code:- 
   SELECT 
    upper(plan) as plan,
    count(customer_id) as sb_cnt
FROM subscriptions
WHERE start_date IS NOT NULL AND start_date <> '' AND LOWER(status) = 'active'
GROUP BY 1
ORDER BY 1 ASC;


• What is the Average Revenue Per User (ARPU)?
Code:- 
    SELECT 
    sum(cast(monthly_price as int))/count(distinct customer_id) as avg_rev_per_users    
FROM subscriptions
WHERE start_date IS NOT NULL AND start_date <> '' AND LOWER(status) = 'active';

DROP TABLE IF EXISTS customers CASCADE;


-- CREATE NEW CUSTOMERS TABLE.
CREATE TABLE customers (
    customer_id VARCHAR(50) PRIMARY KEY,
    customer_name VARCHAR(100),
    email VARCHAR(100),
    dob VARCHAR(50),               
    gender VARCHAR(20),
    country VARCHAR(50),
    signup_date VARCHAR(50),       
    acquisition_channel VARCHAR(100),
    device_type VARCHAR(100),
    preferred_language VARCHAR(50),
    customer_status VARCHAR(50),
    churn_date VARCHAR(50),        
    churn_reason VARCHAR(255)     
);
